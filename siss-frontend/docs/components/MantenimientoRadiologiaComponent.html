export class MantenimientoRadiologiaComponent implements OnInit {
  private radSvc = inject(RadiologiaService);
  private estSvc = inject(EstablecimientosService);
  private notify = inject(NotificationService);

  tab = signal<'catalogo' | 'asignacion'>('catalogo');
  
  estudios   = signal<ExamenRadiologico[]>([]);
  categorias = signal<string[]>([]);
  busqueda   = '';
  categoria  = '';

  establecimientos   = signal<any[]>([]);
  centroSeleccionado = signal<any | null>(null);
  estudiosAsignados  = signal<Set<number>>(new Set());
  guardando          = signal(false);

  ngOnInit() {
    this.cargarCatalogo();
    this.cargarEstablecimientos();
    this.radSvc.listarCategorias().subscribe({
      next: (cats) => this.categorias.set(cats),
      error: (err) => console.error('Error al cargar categorías de rad', err)
    });
  }

  cargarCatalogo() {
    this.radSvc.listarCatalogo({ busqueda: this.busqueda, categoria: this.categoria }).subscribe({
      next: (res: any) => {
        const data = res.data ?? res;
        this.estudios.set(Array.isArray(data) ? data : (data.data ?? []));
      },
      error: (err) => console.error('Error al cargar catálogo rad', err)
    });
  }

  cargarEstablecimientos() {
    this.estSvc.listar().subscribe({
      next: (res: any) => this.establecimientos.set(res.data ?? res),
      error: (err: any) => console.error('Error al cargar establecimientos', err)
    });
  }

  seleccionarCentro(est: any) {
    this.centroSeleccionado.set(est);
    this.estudiosAsignados.set(new Set());
    this.radSvc.obtenerAsignaciones(est.id).subscribe({
      next: (res: any) => {
        const data = res.data ?? res;
        const ids = Array.isArray(data) ? data : (data.data ?? []);
        const newSet = new Set<number>();
        ids.forEach((id: any) => newSet.add(Number(id)));
        this.estudiosAsignados.set(newSet);
      },
      error: (err) => console.error('Error al cargar asignaciones rad', err)
    });
  }

  filtrarPorCat(cat: string) {
    return this.estudios().filter(ex => ex.categoria === cat);
  }

  estaAsignado(id: any) {
    return this.estudiosAsignados().has(Number(id));
  }

  toggleEstudio(id: any) {
    const numId = Number(id);
    this.estudiosAsignados.update(set => {
      const newSet = new Set(set);
      if (newSet.has(numId)) newSet.delete(numId);
      else newSet.add(numId);
      return newSet;
    });
  }

  marcarTodaCategoria(cat: string, valor: boolean) {
    const ids = this.filtrarPorCat(cat).map(ex => Number(ex.id));
    this.estudiosAsignados.update(set => {
      const newSet = new Set(set);
      ids.forEach(id => {
        if (valor) newSet.add(id);
        else newSet.delete(id);
      });
      return newSet;
    });
  }

  guardarAsignacion() {
    const est = this.centroSeleccionado();
    if (!est) return;

    this.guardando.set(true);
    const ids = Array.from(this.estudiosAsignados());
    this.radSvc.asignarEstudios(est.id, ids).subscribe({
      next: () => {
        this.notify.success(`Catálogo de radiología para ${est.nombre} actualizado.`);
        this.guardando.set(false);
      },
      error: (err) => {
        this.notify.error('Fallo al sincronizar el catálogo de radiología.');
        this.guardando.set(false);
      }
    });
  }
}
